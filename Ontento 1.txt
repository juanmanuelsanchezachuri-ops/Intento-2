  <script>
    const celdas = document.querySelectorAll('.celda');
    const textoEstado = document.getElementById('estado');
    const btnReiniciar = document.getElementById('reiniciar');
    const btnResetMarcador = document.getElementById('reset-marcador');
    const puntosXElem = document.getElementById('puntos-x');
    const puntosOElem = document.getElementById('puntos-o');

    let jugadorActual = 'X';
    let estadoJuego = ["", "", "", "", "", "", "", "", ""];
    let juegoActivo = true;
    let puntosX = 0;
    let puntosO = 0;

    const combinacionesGanadoras = [
      [0, 1, 2], [3, 4, 5], [6, 7, 8], // Filas
      [0, 3, 6], [1, 4, 7], [2, 5, 8], // Columnas
      [0, 4, 8], [2, 4, 6]             // Diagonales
    ];

    function manejarClickCelda(evento) {
      const celdaClickeada = evento.target;
      const indice = parseInt(celdaClickeada.getAttribute('data-indice'));

      if (estadoJuego[indice] !== "" || !juegoActivo) {
        return;
      }

      actualizarCelda(celdaClickeada, indice);
      verificarResultado();
    }

    function actualizarCelda(celda, indice) {
      estadoJuego[indice] = jugadorActual;
      celda.textContent = jugadorActual;
      celda.classList.add(jugadorActual.toLowerCase());
    }

    function cambiarJugador() {
      jugadorActual = jugadorActual === 'X' ? 'O' : 'X';
      const nombreJugador = jugadorActual === 'X' ? 'Jugador 1 (X)' : 'Jugador 2 (O)';
      textoEstado.textContent = `Turno de: ${nombreJugador}`;
    }

    function verificarResultado() {
      let victoria = false;

      for (let i = 0; i < combinacionesGanadoras.length; i++) {
        const [a, b, c] = combinacionesGanadoras[i];
        if (estadoJuego[a] === "" || estadoJuego[b] === "" || estadoJuego[c] === "") {
          continue;
        }
        if (estadoJuego[a] === estadoJuego[b] && estadoJuego[b] === estadoJuego[c]) {
          victoria = true;
          break;
        }
      }

      if (victoria) {
        const ganador = jugadorActual === 'X' ? 'Jugador 1 (X)' : 'Jugador 2 (O)';
        textoEstado.textContent = `¡${ganador} ha ganado! 🎉`;
        juegoActivo = false;

        if (jugadorActual === 'X') {
          puntosX++;
          puntosXElem.textContent = puntosX;
        } else {
          puntosO++;
          puntosOElem.textContent = puntosO;
        }
        return;
      }

      const empate = !estadoJuego.includes("");
      if (empate) {
        textoEstado.textContent = '¡Empate!';
        juegoActivo = false;
        return;
      }

      cambiarJugador();
    }

    function reiniciarJuego() {
      jugadorActual = 'X';
      estadoJuego = ["", "", "", "", "", "", "", "", ""];
      juegoActivo = true;
      textoEstado.textContent = `Turno de: Jugador 1 (X)`;

      celdas.forEach(celda => {
        celda.textContent = "";
        celda.classList.remove('x', 'o');
      });
    }

    function reiniciarMarcador() {
      puntosX = 0;
      puntosO = 0;
      puntosXElem.textContent = 0;
      puntosOElem.textContent = 0;
      reiniciarJuego();
    }

    celdas.forEach(celda => celda.addEventListener('click', manejarClickCelda));
    btnReiniciar.addEventListener('click', reiniciarJuego);
    btnResetMarcador.addEventListener('click', reiniciarMarcador);
  </script>
